function generateQuestion(){ if(gameOver || lives <= 0){ endGame(); return; } clearInterval(timer); answered = false; message.textContent = ""; next.classList.remove("show"); options.innerHTML = ""; questionNumber++; let a, b, c; let answer; let text; /* ============================== MEDIUM MATH QUESTIONS ============================== */ let type = random(1, 8); // 1. Addition if(type === 1){ level.textContent = "🟢 Medium • Addition"; a = random(100, 600); b = random(100, 600); answer = a + b; text = `${a} + ${b} = ?`; } // 2. Subtraction else if(type === 2){ level.textContent = "🟢 Medium • Subtraction"; a = random(300, 900); b = random(100, a - 50); answer = a - b; text = `${a} − ${b} = ?`; } // 3. Multiplication else if(type === 3){ level.textContent = "🟡 Medium • Multiplication"; a = random(12, 30); b = random(6, 20); answer = a * b; text = `${a} × ${b} = ?`; } // 4. Division else if(type === 4){ level.textContent = "🟡 Medium • Division"; b = random(5, 20); answer = random(10, 40); a = b * answer; text = `${a} ÷ ${b} = ?`; } // 5. Bigger Addition else if(type === 5){ level.textContent = "🟠 Medium • Challenge"; a = random(200, 900); b = random(100, 700); c = random(50, 300); answer = a + b + c; text = `${a} + ${b} + ${c} = ?`; } // 6. Mixed × and + else if(type === 6){ level.textContent = "🟠 Medium • Mixed"; a = random(10, 30); b = random(5, 15); c = random(10, 50); answer = (a * b) + c; text = `(${a} × ${b}) + ${c} = ?`; } // 7. Mixed − and × else if(type === 7){ level.textContent = "🟠 Medium • Mixed"; a = random(30, 80); b = random(5, 15); c = random(5, 30); answer = (a - b) * c; text = `(${a} − ${b}) × ${c} = ?`; } // 8. Division + Addition else{ level.textContent = "🟠 Medium • Mixed"; b = random(5, 15); c = random(10, 50); answer = random(10, 30); a = b * answer; let finalAnswer = answer + c; correctAnswer = finalAnswer; text = `(${a} ÷ ${b}) + ${c} = ?`; showQuestion(text, finalAnswer); return; } correctAnswer = answer; showQuestion(text, answer); } /* ================================= SHOW QUESTION ================================= */ function showQuestion(text, answer){ question.textContent = text; let answers = [answer]; /* Wrong answers पनि नजिकका राख्ने, त्यसैले प्रश्न धेरै सजिलो हुँदैन। */ while(answers.length < 4){ let difference = random(3, 30); let wrong; if(Math.random() < 0.5){ wrong = answer + difference; }else{ wrong = answer - difference; } if( wrong >= 0 && wrong !== answer && !answers.includes(wrong) ){ answers.push(wrong); } } shuffle(answers); answers.forEach(value => { const button = document.createElement("button"); button.className = "option"; button.textContent = value; button.onclick = () => { selectAnswer( button, value ); }; options.appendChild(button); }); document.getElementById( "questionNo" ).textContent = questionNumber; /* प्रत्येक प्रश्नमा 10 seconds */ startTimer(); }
