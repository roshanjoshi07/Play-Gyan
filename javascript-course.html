21. Async JavaScript

Asynchronous code ले लामो समय लाग्ने कामलाई राम्रोसँग handle गर्न मद्दत गर्छ। 

function loadData(){

    return new Promise(
        function(resolve){

            setTimeout(
                function(){

                    resolve(
                        "Data loaded"
                    );

                },
                1000
            );

        }
    );

}

async function start(){

    const data =
    await loadData();

    console.log(data);

}

start();
